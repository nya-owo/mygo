cmake_minimum_required(VERSION 3.20)
project(cloudy_mtsdf_boundary_revision4 LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(MSDFGEN_CORE_ONLY ON CACHE BOOL "" FORCE)
set(MSDFGEN_USE_VCPKG OFF CACHE BOOL "" FORCE)
set(MSDFGEN_USE_SKIA OFF CACHE BOOL "" FORCE)
set(MSDFGEN_BUILD_STANDALONE OFF CACHE BOOL "" FORCE)
set(MSDFGEN_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(msdfgen)
include(skia-build.cmake)
add_library(cloudy_mtsdf SHARED bridge.cpp resolve-shape-geometry.cpp)
target_compile_definitions(cloudy_mtsdf PRIVATE MSDFGEN_USE_SKIA SK_RELEASE NOMINMAX)
target_compile_features(cloudy_mtsdf PRIVATE cxx_std_20)
target_link_libraries(cloudy_mtsdf PRIVATE skia-pathops msdfgen::msdfgen-core)
